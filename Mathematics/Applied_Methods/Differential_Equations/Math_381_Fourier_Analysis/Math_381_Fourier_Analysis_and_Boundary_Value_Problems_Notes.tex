\section{Partial Differential Equations}
\begin{example}
    Find the function $u(x,y)$ such that $u: \R^2 \mapsto \R $ that satisfies the differential equation $u_{xy}=0$
\end{example}
We can just integrate twice. Once with respect to $x$ and one with respect to $y$. 
\begin{equation*}
    \begin{split}
        u_{xy} &= 0 \\
        \Rightarrow u_x&=c(x) \\
        \Rightarrow u(x,y) &=C(x) + D(y)
    \end{split}
\end{equation*}

\begin{example}
    Find a solution $u(x,y)$ to solve the transport equation 
    \begin{equation*}
        u_x+yu_y=0
    \end{equation*}
\end{example}
Conceptually we can think of this as $\vec{v}\cdot \nabla u = D_{\vec{v}}u=0$ where $\vec{v} = \langle 1,y\rangle$. We effectively want to travel along a path such that the directional derivative in $\vec{v}$ is 0. These paths are called integral curves as they are tangent to the vector field at every point. We now arrive at the characteristic equation 
\begin{equation*}
    \od{y}{x}=\frac{y}{1}
\end{equation*}
since the slope at every point is just rise over run and therefore $\frac{y}{1}$. This leads to the solution 
\begin{equation*}
    y(x) = Ae^{x}
\end{equation*}
where $A$ is found using an initial condition. We note that since the value of the function doesn't depend on where you on each curve but rather \emph{which} curve you're on, we can write $u(x,y) =f(A)$ where $A=ye^{-x}$. Therefore the general solution to the transport equation is 
\begin{equation*}
    u(x,y) = f(ye^{-x})
\end{equation*}
\subsection{Initial and Boundary Conditions}
When we have an $n$-order derivative with respect to one variable in an equation, we then need $n$ conditions in order to uniquely determine a solution to a boundary value problem. For derivatives with respect to time, these are generally deemed initial conditions because they are occur at $t=0$. For variables associated with some sort of domains, these are usually deemed boundary conditions since they occur at the boundary of the system of interest. It is worth noting that some boundary conditions can lead to nonsensical and impossible implications. Let us take the boundary value problem 
\begin{equation*}
    y''+y=x \twen y(0)=y(\pi)=0
\end{equation*}
The solution to the homogeneous equation is $A\cos(x)+B\sin(x)$ and the solution to the particular is $x$. Therefore the general solution is $y_g(x)=A\cos x+B\sin x+x$. Plugging in the boundary values yields to following 
\begin{equation*}
    \begin{split}
        y(0)=0=A\\
        y(\pi)=0=\pi
    \end{split}
\end{equation*}
For PDEs we generally do not have to worry about dimensions of boundary conditions because generally it's a boundary in space and is ``infinite dimensional."
\subsection{Wave Equation}
\subsubsection{Solutions}
\begin{theorem}
    Every solution to the wave equation can be written in the form 
    \begin{equation*}
        u(t,x) = p(\xi)+q(\eta) = p(x-ct)+q(x+ct) \label{theorem:1.2}
    \end{equation*}
\end{theorem}
\begin{proof}
    We change our variables to get 
    \begin{equation*}
        u(t,x) = v(x-ct,x+ct)=v(\xi,\eta)=u\left(\frac{\eta - \xi}{2c}, \frac{\eta+\xi}{2}\right)
    \end{equation*}
    By the chain rule, we get
    \begin{equation}
        \pd{u}{t}=c\left(-\pd{v}{\xi}+\pd{v}{\eta}\right) \hspace{20pt} \pdt{u}{t} = c^2\left(\pdt{v}{\xi} - 2\frac{\partial^2v}{\partial \xi \partial\eta}+\pdt{v}{\eta}\right) \label{eq:1.19}
    \end{equation} % Equation 1.19
    \begin{equation}
        \pd{u}{x} = \pd{v}{\xi}+\pd{v}{\eta} \hspace{20pt} \pdt{u}{x} = \pdt{u}{\xi}+2\frac{\partial^2v}{\partial\xi\partial\eta} + \pdt{v}{\eta} \label{eq:1.20}
    \end{equation} % Equation 1.20
    This implies 
    \begin{equation}
        \square u=-4c^2\frac{\partial^2v}{\partial\xi\partial\eta}=0 \text{ iff } \frac{\partial^2v}{\partial\xi \partial\eta} = 0 \label{eq:1.21}
    \end{equation} % Equation 1.21
    If we create a new variable $w=\pd{v}{\eta}$, we get from the condition in \eqref{eq:1.21} that $\pd{w}{\xi}=0 \therefore w=r(\eta)$. We then get the differential equation 
    \begin{equation*}
        \pd{v}{\eta} = r(\eta) \rightarrow\int\pd{v}{\eta}d\eta = \int r(\eta)d\eta
    \end{equation*}
    By performing the integration with respect to $\eta$, we then get a constant as a function of $\xi$ which leads to the following
    \begin{equation*}
        v(\xi,\eta) - p(\xi)=q(\eta) \rightarrow v(\xi,\eta)=p(\xi)+q(\eta)
    \end{equation*}
\end{proof}

\begin{theorem}[d'Alembert's Formula]
    The solution to the wave equation $u_{tt}=c^2u_{xx}$ with initial conditions $u(0,x)=f(x)$ and $u_t(0,x)=g(x)$ is 
    \begin{equation*}
        u(t,x) = p(\xi)+q(\eta) = \frac{f(\xi)+f(\eta)}{2}+\frac{1}{2c}\int^\eta_\xi g(z)dz 
    \end{equation*}
    where $\xi=x-ct$ and $\eta =x+ct$
\end{theorem}

\begin{proof}
    \begin{equation}
    u(0,x)=p(x-0)+q(x-0)=f(x) \rightarrow p'(x)+q'(x)=f'(x) \label{eq:1.22}
\end{equation} % Equation 1.22
\begin{equation}
    u_t(0,x) = -cp'(x)+cq'(x) = g(x) \rightarrow-p'(x)+q'(x)=\frac{1}{c}g(x) \label{eq:1.23}
\end{equation} % Equation 1.23
If we subtract Equations $\eqref{eq:1.22}$ and $\eqref{eq:1.23}$, we get 
\begin{equation*}
    2p'(x) = f'(x)-\frac{1}{c}g(x) \rightarrow \frac{1}{2}f(x)-\frac{1}{2c}\int^x_0g(z)dz + a 
\end{equation*}
\begin{equation*}
    q(x)=f(x)-p(x)=\frac{1}{2}f(x)+\frac{1}{2c}\int^x_0g(z)dz-a
\end{equation*}
By applying Theorem \ref{theorem:1.2} once again, we reach d'Alembert's solution
\begin{equation}
    u(t,x) = p(\xi)+q(\eta) = \frac{f(\xi)+f(\eta)}{2}+\frac{1}{2c}\int^\eta_\xi g(z)dz \label{eq:dASol}
\end{equation}
\end{proof}



When undergoing external forcing, we have to add an additional forcing term 
\begin{equation*}
    \square u=F(t,x)
\end{equation*}
Let us assume for now that we have homogeneous initial conditions, undergoing the same change of variables above, we get 
\begin{equation*}
    \square u=-4c^2\frac{\partial^2v}{\partial\xi\partial\eta}=F\left(\frac{\eta-\xi}{2c},\frac{\eta+\xi}{2}\right)
\end{equation*}
Integrating both sides with respect to $\eta' /\zeta$ (as a dummy variable) from $\xi$ to $\eta$ gets us
\begin{equation*}
    \pd{v}{\xi}(\xi,\eta)-\pd{v}{\xi}(\xi,\xi) = -\frac{1}{4c^2}\int^\eta_\xi F\left(\frac{\zeta-\eta}{2},\frac{\zeta+\eta}{2}\right)d\zeta
\end{equation*}
Rearranging and doing some algebra for the first partial derivatives in Equations \eqref{eq:1.19} and \eqref{eq:1.20}, we get the following
\begin{equation}
    \pd{v}{\xi}(\xi,\eta)=\frac{1}{2}\pd{u}{x}\left(\frac{\eta-\xi}{2},\frac{\eta+\xi}{2}\right)\label{eq:1.25}
\end{equation}
Which if we plug in $\eta=\xi$ and apply the homogeneous initial conditions, we get 
\begin{equation*}
    \pd{v}{\xi}=-\frac{1}{2c}\pd{u}{t}(0,\xi) + \frac{1}{2}\pd{u}{x}(0,\xi) = 0
\end{equation*}
By the same rearrangement and algebra, we get that $v(\eta,\eta)=0$ and so by integrating 1.25 with respect to $\xi'/\chi$ (once again a dummy variable)
\begin{equation}
    -v(\xi,\eta)=-\frac{1}{4c^2}\int^\eta_\xi\int^\eta_\chi F\left(\frac{\eta-\xi}{2},\frac{\eta+\xi}{2}\right)d\zeta d\chi
\end{equation}
We want to change our integration domain and so we take the following change of variables
\begin{equation*}
    \chi=y-cs \text{ and } \zeta=y+cs
\end{equation*}
which then yields the inequality
\begin{equation*}
    x-ct \leq y-cs \leq y+cs \leq x+ct \rightarrow x-c(t-s)\leq y \leq x+c(t-s)    
\end{equation*}
Effectively what we are doing is changing from the characteristic $\chi\zeta$ domain to the physical integration $tx$ domain. $y$ and $s$ just represent dummy variables that we can integrate $F(s,y)$ over the $tx$ plane. The Jacobian results in the relationship $d\chi d\zeta = 2c dsdy$ which leads to the final result of the particular solution to the forced wave equation to be 
\begin{equation*}
    u(t,x) = \frac{1}{2c}\int \! \! \!  \int_DF(s,y)dsdy=\frac{1}{2c}\int^t_0\int^{x+c(t-s)}_{x-c(t-s)}F(s,y)dyds
\end{equation*}
We can now write the general solution to the classical wave equation as the sum of d'Alembert's solution with the solution above to get 
\begin{equation}
    u(t,x)=\frac{1}{2}[f(x-ct)+f(x+ct)] +\frac{1}{2c}\int^{x+ct}_{x-ct}g(y)dy + \frac{1}{2c}\int^t_0\int^{x+c(t-s)}_{x-c(t-s)}F(s,y)dyds
\end{equation}

\subsubsection{Causality and Energy}
Solutions to the wave equation satisfy causality. That is, a change to the initial conditions can only affect the solution $u(t,x)$ for $t>0$ where $|x-x_0|\leq ct$. In other words, the information only propagates at speed $c$.\\\\
We define the \emph{energy} of a solution $u(t,x)$ as 
\begin{equation*}
    E=\frac{1}{2}\int^\infty_{-\infty}u_t^2+c^2u_x^2dx
\end{equation*}
Let us impose the localizing conditions $u(t,x)\rightarrow 0$ as $x\rightarrow \pm \infty$. If we take the time derivative and differentiate under the integral sign
\begin{equation*}
    \begin{split}
        \od{E}{t} &=\int^\infty_{-\infty}u_{tt}u_t+c^2u_xu_{xt}dx \\
        &=\int^\infty_{-\infty} u_tc^2u_{xx}+c^2u_xu_{xt}dx \\
        &=c^2\int^\infty_{-\infty}-u_xu_{xt}+u_xu_{xt}dx \\
        &=0
    \end{split}
\end{equation*}
where we integrated by parts in the middle. This shows that energy is conserved within solutions to the wave equation. 

\subsection{Heat Equation}
\subsubsection{The Maximum Principle}
\begin{theorem}[The Weak Maximum Principle]
    Let $\gamma >0$. Suppose $u(t,x)$ is a solution to the forced heat equation 
    \begin{equation*}
        \pd{u}{t} = \gamma \pdt{u}{x} + F(t,x)
    \end{equation*}
    on the rectangular domain 
    \begin{equation*}
        R=\{a\leq x \leq b, 0 \leq t \leq c \}
    \end{equation*}
    Assuming that the forcing term is nowhere positive, $F(t,x) \leq 0$, then the maximum of $u(t,x)$ on the closed rectangle $\bar{R}$ is attained at $t=0, x=a,$ or $x=b$.
\end{theorem}
\begin{proof}
    Let us first prove the result under the stronger assumption that $F(t,x) = \pd{u}{t} -\gamma \pdt{u}{x} <0$ which implies $\pd{u}{t} < \gamma \pdt{u}{x}$ everywhere in $R$. Suppose there exists a point $(t^\star, x^\star) \in R$ such that $u(t,x)$ is at a local maximum. By definition, its gradient must be 0.
    \begin{equation*}
        \nabla u(t^\star, x^\star ) = \bold{0} \rightarrow u_t(t^\star,x^\star) = u_{x}(t^\star, x^\star) = 0
    \end{equation*}
    Now let us assume a scalar function $h(x) = (t_\star, x)$. $h(x)$ has a maximum at $x=x^\star$ which means by the second derivative test,
    \begin{equation*}
        h''(x^\star) = u_{xx}(t^\star, x^\star) \leq 0
    \end{equation*}
    This is impossible since we showed earlier that $u_t < u_{xx}$ but $u_t = 0$ and $u_{xx} \leq 0$. As such, our initial assumption that there exists a maximum within $R$ is false. Now we need to exclude the possibility of the maximum occurring at the right edge of the rectangle at a point $(t^\star,x^\star) = (c,x^\star)$ where $a<x^\star < b$. If this is the case, then $g(t) =u(t,x^\star)$ is nondecreasing with respect to time at $t=c$ which implies $g'(t) = u_t(t,x^\star) \geq 0$ which is once again with the conditions we established earlier. \\\\
    To generalize our results to the $F(t,x) \leq 0$, we start with a solution to the forced heat equation $u(t,x)$ and add a little perturbation.
    \begin{equation*}
        v(t,x) = u(t,x) + \varepsilon x^2 \text{ where } \varepsilon>0
    \end{equation*}
    By the chain rule, we get the following 
    \begin{equation*}
        \pd{v}{t} = \pd{u}{t} =\gamma \pdt{u}{x}+F(t,x) = \gamma \pdt{v}{x} -2\gamma \varepsilon+F(t,x) = \gamma \pdt{v}{x} +\tilde{F}(t,x)
    \end{equation*}
    Since $F(t,x) \leq 0$ and $\varepsilon>0$, we know that 
    \begin{equation*}
        \tilde{F}(t,x) = F(t,x) -2\gamma\varepsilon <0
    \end{equation*}
    and we can use what we proved previously to state that $v(t,x)$ cannot attain a a maximum anywhere in $R$ or on the right edge of $R$. Now to rigorously show that this applies to $u(t,x)$, let us denote the maximum value of $u(t,x)$ on any of the three sides as $M$. By our previous logic and mimicking the form of $v(t,x)$, we get
    \begin{equation*}
    u(t,x) \leq v(t,x) \leq M+\varepsilon \max \{a^2,b^2 \}.
    \end{equation*}
    Now let $\varepsilon\rightarrow 0^+$ and we have $u(t,x) \leq M$ everywhere in $R$.
    
\end{proof}
This is known as the \textit{Weak Maximum Principle}. The \textit{Strong Maximum Principle} states that given a non-constant solution $u(t,x)$, any non-initial, non-bonudary point $(t,x) \in \hat{R} = \{a<x<b, 0<t\leq c\}$ is strictly less than its maximum initial and boundary values. In math terms, $u(t,x)<M \ \forall(t,x)\in \hat{R}$. Since the Maximum Principle is used to establish Corollary \ref{coro:mM}, the inequalities become strict in there as well. 
\begin{corollary}
    Supposed $u(t,x)$ solves the heat equation $u_t = \gamma u_{xx}$ with $\gamma > 0$ for $a,x<b$ and $0<t<c$. Set 
    \begin{equation*}
        B=\{(0,x) \ | \ a\leq x \leq b\} \cup \{(t,a)\ | \ 0\leq t \leq c\} \cup \{(t,b)\ | \ 0 \leq t \leq c\}
    \end{equation*}
    and let 
    \begin{equation*}
        M = \max \{u(t,x) \ | \ (t,x) \in B\} \hspace{20pt} m=\min \{u(t,x) \ | \ (t,x) \in B \}
    \end{equation*}
    be respectively the maximum and minimum values for the initial and boundary temperatures. Then $m \leq u(t,x) \leq M$ for all $a \leq x \leq b, 0 \leq t \leq c$.\label{coro:mM}
\end{corollary}
\begin{proof}
        The upper bound $u(t,x) \leq M$ follows immediately from the Maximum principle. To establish the lower bound, let us define a new function $\tilde{u}(t,x) = -u(t,x)$ that also solves the heat equation. This satisfies $\tilde{u}(t,x) \leq -m$ on $B$ by the Maximum Principle (in essence what we're doing is flipping the sign such that the minimum now becomes the maximum). This implies that $u(t,x) = -\tilde{u}(t,x) \geq m$.
    \end{proof}
\begin{theorem}[Uniqueness]
    There is at most, one solution to the Dirichlet initial-boundary value problem for the forced heat equation.
\end{theorem}
\begin{proof}
    Suppose $u$ and $\tilde{u}$ are any two solutions to the forced heat equation with the same initial and boundary value conditions. Their difference $v=u-\tilde{u}$ then solves the homogeneous initial-boundary value problem for the unforced heat equation (since the forcing terms cancel out). By the Maximum Principle and Corollary \ref{coro:mM}, we get the maximum and minimum must appear on the boundaries and therefore $m=0\leq v(t,x) \leq M=0$ everywhere which implies $u\equiv \tilde{u}$ and therefore $u$ is unique.
\end{proof}
We can also use an energy method to determine the uniqueness of the solutions. Suppose $u,v$ solve the heat equation and $u=v$ on $\partial R$. We can define a $w=u-v$ and then $w=0$ on $\partial R$. We now define some quantity 
\begin{equation*}
    E(t)=\int^L_0\frac{1}{2}w^2(x,t)dt
\end{equation*}
If we take time derivatives, we get 
\begin{equation*}
    \begin{split}
        \od{E}{t} &=\int^L_0ww_tdt \\
        &=\int^L_0 wkw_{xx}dx \\
        &=kww_x\bigg|^L_0 - k\int^L_0w_x^2dx\\
        &=-k\int^L_0w_x^2dx
    \end{split}
\end{equation*}
If we set the time equal to zero, we see that hte energy is 
\begin{equation*}
    E(0)=\int^L_0\frac{1}{2}w(x,0)^2dx =0
\end{equation*}
by the fact that $w=0$ on $\partial R$. However, $E(t)$ is non-increasing for $t\in [0,T]$ and since energy cannot be negative, $w(x,t)=0$ in $R$. 
\subsubsection{The Fundamental Solution to the Heat Equation}
Suppose $u(t,x)$ solves the heat equation. Some special properties of these solutions are that they are space and time invariant, there are some special symmetry and similarity solutions, its derivatives solve the heat equation, and linear combinations also solve the heat equation. We therefore seek a solution $Q(t,x)$ with initial condition $Q(0,x)=H(x)$. We choose this form of a solution since $Q(0,ax)=Q(0,x)\,\,\, \forall a>0$. Let us say $Q(x,t)=g(p) $ where $p=\frac{x}{\sqrt{4kt}}$. Therefore 
\begin{equation*}
    \begin{split}
        Q_t &=-\frac{x}{2\sqrt{4k}}xt^{-3/2}g'(p) \\
        Q_{xx}&= g''(p)\left(\frac{1}{\sqrt{4k}}t^{-1/2}\right)^2
    \end{split}
\end{equation*}
and therefore the heat equation turns into 
\begin{equation*}
    -g'(p)\frac{x}{2\sqrt{4k}}xt^{-3/2} = kg''(p)\frac{1}{4k}t^{-1}
\end{equation*}
and therefore we get 
\begin{equation*}
    g''(p)+g'(p)2p=0
\end{equation*}
If we let $h(p)=g'(p)$, we can see that this is a linear first order differential equation and we using an integrating factor $e^{p^2}$. This leads us to get that 
\begin{equation*}
    h=g'(p)=Ae^{-p^2}
\end{equation*}
Therefore 
\begin{equation}
    Q(t,x)=g(p)=A\int^p_{-\infty} e^{-q^2}dq+B
\end{equation}
If $x>0$ and $t\rightarrow 0^+$, then $Q(t,x)$ we have that 
\begin{equation}
    Q(t,x)=A\sqrt{\pi}=1 \to A=\frac{1}{\sqrt\pi}
\end{equation}
If we have some initial condition $\phi (x)$, we can impose some localizing boundary conditions as $x\to \pm \infty$. We can write $\phi(x)$ in the following form 
\begin{equation*}
    \begin{split}
        \phi(x)&=\int^x_{-\infty}\phi'(y)dy \\&= \int^\infty_{-\infty}Q(0,x-y) \phi '(y)dy \\
        &= - \int\phi(y)Q_x(0,x-y) (-1)dy \\
        &=\int^\infty_{-\infty}\phi (y)Q_x(0,x-y)dy
    \end{split}
\end{equation*}
where the $Q_x$ derivative implies with respect to the second entry and not $x$ explicitly inside of the argument. From above, we can wee that 
\begin{equation*}
    Q_x(t,x) = \frac{1}{\sqrt {4\pi kt}}e^{-x^2/4kt}
\end{equation*}
and therefore 
\begin{equation*}
    \phi (x)=\frac{1}{\sqrt{4\pi kt}}\int^\infty_{-\infty} \phi(y)e^{-\frac{(x-y)^2}{4kt}}dy
\end{equation*}
\subsection{Separation of Variables}
Let us suppose we are given the equation 
\begin{equation*}
    u_t=ku_{xx} \twen x\in [0,l] \twen t\geq 0
\end{equation*}
with $u(0,t)=u(l,t)=0 \,\,\, \forall t$ and $u(0,x)=\phi(x)$. We start with the ansatz $u(t,x)=T(t)X(x)$. Substituting this into the heat equation, we have 
\begin{equation*}
    T'X=kTX'' 
\end{equation*}
and dividing both sides by $XT$, we get 
\begin{equation*}
    \frac{T'}{kT} = \frac{X''}{X} = \lambda
\end{equation*}
For the temporal equation, we get that 
\begin{equation*}
    \frac{T'}{kT}=\lambda \to T=Ae^{\lambda kt}
\end{equation*}
and for the spatial we have 
\begin{equation*}
    \frac{X''}{X}=\lambda
\end{equation*}
We see with here that we require $\lambda <0$ because if $\lambda >0$, we would get $\sinh$ and $\cosh$ spatial solutions which can't satisfy the boundary conditions. $\lambda \neq 0$ because we wold get a linear eigenfunction which can't generate an arbitrary function. The boundary conditions also imply that the linear function would be the trivial solution and therefore 0 can't be an eigenvalue because of that. As such, let $\lambda = -\alpha^2$. With these, we get that $X=B\cos(\alpha x)+C\sin(\alpha x)$. From the boundary values, we see that $B=0$ and $\alpha l = n\pi$ and therefore $\alpha = \frac{n\pi}{l}$. Combining everything, we get 
\begin{equation*}
    u(t,x)=\sum^\infty_{n=1}e^{-\alpha_n^2kt}b_n\sin(\alpha_nx)
\end{equation*}
and thus the initial condition 
\begin{equation*}
    \phi(x) = \sum^\infty_{n=1}b_n\sin (\alpha_nx)
\end{equation*}
This is the genesis of Fourier Analysis.

\section{Fourier Series}
We saw in the previous section that any arbitrary function $\phi(x)$ can be represented as 
\begin{equation*}
    \phi (x) = \sum^\infty_{n=1} A_n \sin\left(\frac{n\pi x}{l}\right)
\end{equation*}
We also have a key identity that 
\begin{equation*}
    \int^l_0 \sin \frac{n\pi x}{l} \sin \frac{m\pi x}{l} dx = \frac{l}{2}\delta_{n,m}
\end{equation*}
where $\delta_{n,m}$ is the Kronecker delta and is 0 when $n\neq m$ and 1 when $n=m$. If we take the integral of $\phi$ and its series representation, we get 
\begin{equation*}
    \begin{split}
        \int^l_0 \phi(x)\sin \frac{m\pi x}{l}dx &= \sum^\infty_{n=1}A_n \int^l_0 \sin \frac{n\pi x}{l}\sin \frac{m\pi x}{l}dx\\ &= A_m \frac{l}{2} \\
        \Rightarrow A_m&=\frac{2}{l}\int^l_0 \phi(x)\sin \frac{m\pi x}{l}dx
    \end{split}
\end{equation*}
\subsection{Sturm-Liouville Theory}
Let us consider $A$ to be the differential operator $-\odt{}{x}$ which acts on all smooth function on the interval $(a,b)$. To see if it's a symmetric operator, we want to see if $\langle Af, g\rangle = \langle f,Ag\rangle $. This is analogous to symmetric matrices which we see 
\begin{equation*}
    \begin{split}
        \ip{Mx}{y} &= (Mx)^Ty \\
        &=x^TM^Ty \\
        &=\ip{x}{My}\,\,\, \forall x,y
    \end{split}
\end{equation*}
We start by just computing it using the $L^2$ inner product
\begin{equation*}
    \begin{split}
        \ip{Af}{g}&=\int^b_a -\odt{}{x}fgdx \\
        &=-\od{f}{x}g\Bigg|^b_a +\int^b_a \od{f}{x}\od{g}{x}dx \\
        &=-\od{f}{x}g\Bigg|^b_a +f\od{g}{x} -\int^b_af \odt{}{x}gdx \\
        &=\ip{f}{Ag} + \left[-f'(b)g(b)+f(b)g'(b)+f'(a)g(a)-f(a)g'(a) \right]
    \end{split}
\end{equation*}
where everything in brackets we shall denote as $B$. If we want $A$ to be a symmetric operator, we require that $B=0$. Under various boundary conditions, $B=0$ and these would be homogeneous Dirichlet, homogeneous Neumann, and periodic boundary conditions. 
\begin{definition}
    Boundary conditions are \emph{symmetric} if $B=0$ for all smooth functions $f(x)$ and $g(x)$ on $(a,b)$.
\end{definition}
This means we need 2 linear combinations of boundary conditions that vanish. 
\begin{theorem}
    Suppose we impose symmetric boundary conditions. If $f,g$ are eigenfunctions of some operator $A$ with different eigenvalues,  $\lambda_f \neq \lambda _g$, then $\ip{f}{g} =0$.
\end{theorem}
\begin{proof}
    \begin{equation*}
        \begin{split}
            \ip{Af}{g} &=\ip{\lambda_ff}{g} \\
            &= \lambda_f \ip{f}{g}\\
        \end{split}
    \end{equation*}
    but also $\ip{Af}{g} =\ip{f}{Ag}=\lambda_g\ip{f}{g}$ and therefore we have the equality 
    \begin{equation*}
        \lambda_f \ip{f}{g} = \lambda_g\ip{f}{g}
    \end{equation*}
    Since $\lambda_f\neq \lambda_f$, this equality can only be true if $\ip{f}{g}$. 
\end{proof}
\subsection{Generalized Fourier Series}
Suppose $\phi(x)$ can be written in the form 
\begin{equation*}
    \phi(x) = \sum^\infty_n A_n X_n(x)
\end{equation*}
where $X_n(x)$ are distinct eigenfunctions with distinct eigenvalues. Then 
\begin{equation*}
    \begin{split}
        \ip{\phi}{X_m} &= \ip{ \sum^\infty_n A_nX_n(x)}{X_n} \\
        &=A_m\ip{X_m}{X_m} \\
        \Rightarrow A_m&= \frac{\ip{\phi}{X_m}}{\ip{X_m}{X_m}}
    \end{split}
\end{equation*}
These eigenvaules must also be real. Let us take a look as to why. If $f,g:(a,b)\mapsto \mathbb{C}$, we define the inner product to be 
\begin{equation*}
    \ip{f}{g} =\int^b_af(x)\overline{g(x)}dx
\end{equation*}
We then have $\ip{Af}{g} =\ip{f}{Ag}$ if 
\begin{equation*}
    f'\bar g - f\bar g' \Big|^b_a=0
\end{equation*}
\begin{theorem}
    If we have symmetric boundary conditions, then all eigenvalues are real.
\end{theorem}
\begin{proof}
    If $Af=\lambda f$, then $\ip{Af}{f}=\lambda \ip{f}{f}$ but also $ \ip{Af}{f} =\ip{f}{Af}=\bar \lambda \ip{f}{f}$. Therefore we require that $\lambda =\bar \lambda$ which is only true when $\lambda $ is real.
\end{proof}
If we suppose $X_1,X_2,\ldots$ are pairwise orthogonal eigenfunctions with eigenvalues $\lambda_1,\lambda_2,\ldots$, then we can arrange them in such a way that $\lambda_1 \leq \lambda_2\leq \ldots$. Suppose we have a function $f:(a,b)\mapsto \R$, we can write its generalized Fourier Series as 
\begin{equation*}
    f(x) =\sum^\infty_{n=1} \frac{\ip{f}{X_n}}{\ip{X_n}{X_n}} X_n
\end{equation*}
Our question is, in what sense can this generalized Fourier Series equal $f(x)$? To understand this, we need to understand some notions of convergence.
\begin{definition}[Pointwise Convergence]
    A series $\displaystyle{\sum^\infty_{n=1}g_n(x)}$ converges \emph{pointwise} to $f$ on $(a,b)$ if for all $x\in (a,b)$, we have 
    \begin{equation*}
        \left| f(x)-\sum^N_{n=1}g_n(x) \right| \rightarrow 0 \text{ as } N\to \infty
    \end{equation*}
\end{definition}
\begin{definition}[$L^2$ Convergence]
    A series $\displaystyle{\sum^\infty_{n=1}g_n(x)}$ \emph{converges in $L^2$} (or \emph{mean squared convergence} if 
    \begin{equation*}
        \bigintsss
        ^b_{\!\!\!\!\!\!\!\!a}\,\,\,\left|f(x)-\sum^N_{n=1}g_n(x)\right|^2dx \to 0 \text{ as } N\to \infty
    \end{equation*}
\end{definition}
In a sense, pointwise convergence allows for slow convergence of some points. $L^2$ convergence allows for there to be big errors but over intervals with small measures such that the integral smooths over it.\\\\
If we assume $X_n$ are orthonormal, then we have 
\begin{equation*}
    f = \sum^\infty_{n=1}\ip{f}{X_n}X_n
\end{equation*}
If we let the partial sums be denoted as $S_N(x)=\displaystyle{\sum^N_{n=1}\ip{f}{X_n}{X_n}}$, we then calculate their differences with $f$ as 
\begin{equation*}
    \begin{split}
        ||f-S_n||^2 &= \ip{f-S_N}{f-S_N} \\
        &= \ip{f}{f}-2\ip{f}{S_N}+\ip{S_N}{S_N} \\
        &= ||f||^2-2\sum^N_{n=1}\ip{f}{X_n}^2 + \ip{\sum^N_{n=1}\ip{f}{X_n}X_n}{\sum^N_{k=1}\ip{f}{X_k}X_k} \\
        &= ||f||^2-2\sum^N_{n=1} \ip{f}{X_n}^2+\sum^N_{n=1}\ip{f}{X_n}^2 \\
        &=||f||^2-\sum^N_{n=1}\ip{f}{X_n}^2
    \end{split}
\end{equation*}
There are three important consequences from this calculation:
\begin{enumerate}
    \item Bessel's Inequality
    \begin{equation*}
        \sum^N_{n=1} \ip{f}{X_n}^2 \leq ||f||^2
    \end{equation*}
    which effecitvely says this series will always converge since it is bounded above.
    \item Parseval's Equality which states that $S_N(x)$ converges to $f(x)$ in $L^2$ if and only 
    \begin{equation*}
        \sum^\infty_{n=1}\ip{f}{X_n}^2 = ||f||^2
    \end{equation*}
    \item Riemann-Lebesgue Lemma 
    \begin{equation*}
        \lim_{n\to \infty} \ip{f}{X_n} \to 0
    \end{equation*}
\end{enumerate}
These results are under the assumption that $f(x)$ is square integrable. We can state a few theorems.
\begin{theorem}
    $S_N(x)$ converges to $f(x)$ in $L^2$ for any symmetric boundary conditions.
\end{theorem}
\begin{theorem}[Dirichlet's Theorem]
    For classical Fourier series, consider the function periodic function $f:\R \mapsto \R$. Suppose the partial sums are described by 
    \begin{equation*}
        S_N(x) = \frac{A_0}{2} + \sum^N_{n=1}(A_n\cos(nx)+B_n\sin (nx))
    \end{equation*}
    If $f$ and $f'$ are piecewise continuous functions, then $\displaystyle{\lim_{N\to \infty}S_N(x)}$ converges pointwise for all $x$ and at the discontinuities, we have that 
    \begin{equation*}
        \lim_{N\to\infty}S_N(x)=\frac{1}{2}(f(x_-)+f(x_+))
    \end{equation*}
\end{theorem}
\section{More Boundary Value Problems}
\subsection{Non-Homogeneous Boundary Conditions}
Consider the boundary value problem 
\begin{equation*}
    u_t=ku_{xx} \twen x\in [0,l] \twen t\geq 0
\end{equation*}
with initial and boundary conditions as 
\begin{equation*}
    u(0,x)=0 \twen u(t,0)=h(t) \twen u(l,t)=j(t)
\end{equation*}
Separation of variables is bad here since there doesn't seem to be a good way to deal with the boundary conditions. Let us consider the ansatz 
\begin{equation*}
    u(t,x) = \sum^\infty_{n=1}u_n(t) \sin \frac{n\pi x}{l}
\end{equation*}
where effectively we want to find the Fourier series of the heat distribution for every snapshot of $t$. The wrong approach is plug this directly into the heat equation and get 
\begin{equation*}
    \sum^\infty_{n=1} u'_n(t)\sin \frac{n\pi x}{l} = -k\sum^\infty_{n=1}\left( \frac{n\pi}{l}\right)^2u_n(t)\sin \frac{n\pi x}{l}
\end{equation*}
This is wrong because there are no notions of the boundary conditions and also differentiation is inherently a dangerous operation which may lead to blowing up for large $n$ or divergence from the true function for all $x$. What we want to do is to subtract a function with the same boundary conditions to get homogeneous boundary conditions. Let $\mathcal{U}(t,x) = \left(1-\frac{x}{l}\right) h(t) + \frac{x}{l}j(t)$ such that we have $v(t,x)=u(t,x)-\mathcal{U}(t,x)$. By the chain rule, we have that $u_t = v_t =\mathcal{U}_t$, $u_{xx}=v_{xx}+\mathcal{U}_{xx}$ but $\mathcal{U}_{xx} = 0$ since it is linear. As such, our new function $v(t,x)$ satisfies 
\begin{equation*}
    v_t = kv_{xx}-\mathcal{U}_t
\end{equation*}
We now want to a $v(t,x)$ of the form 
\begin{equation*}
    v(t,x)=\sum^\infty_{n=1}v_n(t)\sin\frac{n\pi x}{l}
\end{equation*}
and wo our derivatives are 
\begin{equation*}
    \begin{split}
        v_t &= \sum^\infty_{n=1}v'_n(t)\sin \frac{n\pi x}{l} \\
        v_{xx} &= -\left( \frac{n\pi}{x} \right)^2 \sum^\infty_{n=1}v_n(t)\sin \frac{n\pi x}{l}
    \end{split}
\end{equation*}
We can say this because suppose it wasn't of this form but rather 
\begin{equation*}
    v_{xx}=\sum^\infty_{n=1}w_n(t)\sin \frac{n\pi x}{l}
\end{equation*}
Then we must have 
\begin{equation*}
    w_n(t) = \frac{2}{l}\int^l_0v_{xx}\sin \frac{n\pi x}{l}dx
\end{equation*}
and due to symmetric boundary conditions, we have that 
\begin{equation*}
    \ip{v_xx}{\sin\frac{n\pi x}{l}} = \ip{v}{-\left( \frac{n\pi}{l}\right)^2\sin \frac{n\pi x}{l}}
\end{equation*}
and so thus 
\begin{equation*}
    w_n(t) = -\left( \frac{n\pi}{l}\right)^2v_n(t)
\end{equation*}
If we plug this into the heat equation, along with the ansatz 
\begin{equation*}
    \mathcal{U}_t (t,x) = \sum ^\infty_{n=1} \mathcal{U}_n(t) \sin \frac{n\pi x}{l}
\end{equation*}
we get 
\begin{equation*}
    \begin{split}
        \sum^\infty_{n=1}v'_n(t) \sin \frac{n\pi x}{l} &=  \sum^\infty_{n=1} \left( - \left( \frac{n\pi }{l}\right)^2 k v_n(t)- \mathcal{U}_n(t) \right) \sin \frac{n\pi x}{l} \\
        v_n'(t) &= -\left( \frac{n\pi }{l} \right)^2kv_n(t)- \mathcal{U}_n(t) \\
        v_n(t)&= C_ne^{(\frac{n\pi}{l})^2kt } - \int^t_0e^{-(\frac{n \pi}{l})^2k(t-s)} \mathcal{U}_n(s)ds
    \end{split}
\end{equation*}
Once we figure out the $v_n$'s, we can therefore reconstruct 
\begin{equation*}
    v(t,x) = \sum^\infty_{n=1}v_n(t) \sin \frac{n\pi x}{l}
\end{equation*}
and we can find the coefficients using the initial conditions such that 
\begin{equation*}
    v(0,x)=-\mathcal{U}(0,x) = \sum^\infty_{n=1}C_n \sin \frac{n\pi x}{l}
\end{equation*}
and then reconstruct the original solution using $u(t,x)=v(t,x)+\mathcal{U}(t,x)$.

\subsection{Heat and Wave Equation in $\R^2,\R^3$}
Let us consider a region $D \subset \R^2, \R^3$ and we want to solve either the heat equation 
\begin{equation*}
    u_t = k -\lap u \text{ or } u_{tt}=c^2\lap u
\end{equation*}
Let us consider the example wave equation in $\R^2$ with homogeneous Dirichlet conditions. If we separate variables using the ansatz $u(t,x,y) = X(x)Y(y)T(t)$, then this would be very inconvenient to deal with irregularly shaped boundaries. That is why the correct ansatz is $u(t,x,y) = X(x,y)T(t)$. We now separate as normal 
\begin{equation*}
    \begin{split}
        XT'' &= \lap XT \\
        \frac{T''}{T}=&\frac{\lap X}{X} =-\lambda \\
        \lap  X+\lambda X =& 0 \twen T''+\lambda T=0
    \end{split}
\end{equation*}
We assume that $X: D \mapsto \R$ and satisfies homogeneous Dirichlet conditions. We can now apply our results from Sturm-Liouville Theory with an inner product of 
\begin{equation*}
    \ip{f}{g} =\int \!\!\!\int_D fg dA
\end{equation*}
If it is symmetric, then we expect $\ip{\lap f}{g} = \ip{f}{\lap g}$. For the temporal function, we get that 
\begin{equation*}
    T_n = A_n \cos \left(\sqrt{\lambda_n} t\right) +B_n \sin \left( \sqrt{\lambda_n} t\right)
\end{equation*}
and therefore our solution is 
\begin{equation*}
    u(t,x,y) = \sum^\infty_{n=1} \left( A_n \cos \left(\sqrt{\lambda_n t} \right) +B_n \sin \left(\sqrt{\lambda_n} t\right) \right) X_n
\end{equation*}
We can consider these equations to physically describe say a drumhead of shape $D$ which vibrates at the frequencies $\frac{\sqrt{\lambda_n}}{2\pi}$ where $\lambda_n$ are the eigenvalues of the 2D Laplacian on $D$. One famous question is whether you can ``Hear the Shape of a Drum." In other words, can you determine $D$ from the eigenvalues of the Laplacian over $D$. In general, the answer is no as one can construct two shapes of equal perimeter and area out of triangles that have the same eigenvalues. However, the question about whether there exists a counterexample smooth boundaries is still an open problem. 
\subsection{Laplace's Equation}
Laplace's Equation is given by 
\begin{equation*}
    \lap u = 0
\end{equation*}
where $\lap$ denotes the Laplacian. Functions that satisfy Laplace's equation are deemed \emph{harmonic}. The inhomogeneous Laplace's equation, $-\lap u=f$, is known as the Poisson equation and is one of the most important equations in physics. One interesting application is with Brownian motion. Consider a particle moving in a container $D$ randomly until it hits a side. Let us divide the boundary $\partial D$ into two sections, $C_1$ and $C_2$. Let $u(x,y,z)$ be the probability density function that if a particles starts at $(x,y,z)$, it will hit $C_1$ first. Then we can say that $u$ is harmonic in $D$ and has the boundary conditions $u=1$ on $C_1$ and $u=0$ on $C_2$. We can also state a maximum principle for Laplace's equation. Suppose $\lap u=0$ on a connected domain $D$ such that $u$ extends continuously to $\partial D$. Then the maximum and minimum of $u$ must occur on the boundary. \\\\
Let us consider $\lap u=0$ on the rectangle $[0,a]\times [0,b]$ in $\R^2$ with initial and boundary conditions $u(x,b)=0,u_y(x,0)=0, u(0,y)=0,u_x(a,y)=j(y)$. We want a separated solution $u(x,y)=X(x)Y(y)$ and therefore 
\begin{equation*}
    \frac{X''}{X} = - \frac{Y''}{Y} =\lambda
\end{equation*}
and so our associated eigenvalue problems are $X'' = \lambda X $ and $Y''=-\lambda Y$. Our boundary values implie that $x(0)=y(b)=y'(0)=0$ while the $u_x(a,y)=j(y)$ boundary condition is useless. Solving the $Y$ equation, we have 
\begin{equation*}
    Y_{2n+1}(y) = \cos \left(\frac{(2n+1)\pi y}{2b}\right)
\end{equation*}
where our eigenvalues are $\lambda_n = \left(\frac{n\pi}{2b} \right)^2$. Therefore the $X$ eigenfunctions are 
\begin{equation*}
    X_n(x) = A_n \cosh \left(\sqrt{\lambda_n}x\right)+B_n\sinh \left( \sqrt{\lambda_n}x\right)
\end{equation*}
The $X_n(0)$ boundary conditions imply that $A=0$. Therefore our combined eigenfunctions are 
\begin{equation*}
    u_{2n+1}(x,y)= \sinh \left( \frac{(2n+1)\pi x}{2b}\right) \cos \left( \frac{(2n+1)\pi y}{2b} \right)
\end{equation*}
and so the fully summed up solution is 
\begin{equation*}
    u(x,y) = \sum^\infty_{n=0} B_n\sinh \left( \frac{(2n+1)\pi x}{2b} \right) \cos\left( \frac{(2n+1)\pi y}{2b} \right)
\end{equation*}
We now use the final condition such that 
\begin{equation*}
    \begin{split}
        u_x(x,y) &= \sum^\infty_{n=0} B_n \left(\frac{(2n+1)\pi}{2b} \right) \cosh \left( \frac{(2n+1)\pi x}{2b} \right) \cos\left( \frac{(2n+1)\pi y}{2b} \right) \\
        j(y)=u_x(a,y)&=\sum^\infty_{n=0} B_n \left(\frac{(2n+1)\pi}{2b} \right) \cosh \left( \frac{(2n+1)\pi a}{2b} \right) \cos\left( \frac{(2n+1)\pi y}{2b} \right) 
    \end{split}
\end{equation*}
We can calculate the combined Fourier coefficient as 
\begin{equation*}
    \tilde{B_n} = B_n\left( \frac{n\pi}{2b} \right) \cosh \left( \frac{(2n+1)\pi a}{2b} \right) = \frac{\left\langle j(y),\cos \left( \frac{n\pi y}{2b} \right) \right\rangle }{\left\langle cos \left( \frac{n\pi y}{2b} \right),\cos \left( \frac{n\pi y}{2b} \right) \right\rangle}
\end{equation*}
\subsubsection{Poisson Integral Formula}
Let us solve Laplace's equation $\lap u=0$ on the disk of radius $a$ $D_a \subset \R^2$. $u$ on $\partial D_a = h(\theta)$ for $\theta \in [0,2\pi]$. We also apply a $2\pi$-periodic condition to the $\theta$ part. In polar coordinates, $D_a$ becomes a rectangle. We split $u(r,\theta) = R(r) \Theta (\theta)$. The Laplacian in polar coordinates is $\lap u=u_{rr} + \frac{1}{r}u_r+\frac{1}{r^2} u_{\theta \theta}$ so our equation becomes 
\begin{equation*}
    \begin{split}
        R'' \Theta + \frac{1}{r}R'\Theta + \frac{1}{r^2} R\Theta '' = 0\\ 
        \frac{r^2R''}{R} + \frac{rR'}{R} +\frac{\Theta ''}{\theta} =0 \\
        \frac{r^2R''+R'}{R} = -\frac{\Theta ''}{\Theta}=\lambda
    \end{split}
\end{equation*}
We then get that $\Theta'' = -\lambda \theta$ with periodic boundary conditions and so our eigenfunctions are 
\begin{equation*}
    \Theta _n(\theta) =A_n\cos (n\theta) +B\sin (n\theta)
\end{equation*}
The $R$ equation is $r^2 R''+rR' -\lambda R=0$ which is an Euler Equation and we have the ansatz that $R(r) = r^\alpha$. We then get that 
\begin{equation*}
    \begin{split}
        \alpha(\alpha-1)+\alpha - n^2 =0 \\
        \alpha ^2 -n^2=0 \\
        \alpha = \pm n
    \end{split} 
\end{equation*}
For $|n|>0$, then $R(r) = Cr^n+Dr^{-n}$. If $n=0$, then $R(r) = C+D\ln r$. We need to be careful because at $r=0$, the function loses injectivity and so we impose the condition that $R(r)$ is bounded as $r\to 0$. Because of this, $D =0$ for both cases. For $n=0$, then $u(r,\theta) =A_0$ and for $n>0$, we have $u(r, \theta) = r^n(C\cos (n\theta)+D\sin(n\theta))$ and so our general solution is 
\begin{equation*}
    u(r,\theta) = \frac{1}{2}A_0 +\sum^\infty_{n=1}r^n(A_n\cos(n\theta) + B_n \sin (n\theta))
\end{equation*}
We now impose the boundary condition at $r=a$.
\begin{equation*}
    h(\theta) = \frac{1}{2}A_0 + \sum^\infty_{n=1} a^n ( A_n\cos (n\theta) +B_n \sin (n\theta))
\end{equation*}
Through some clever identities, we can write this as 
\begin{equation*}
    u(r,\theta) = \frac{a^2-r^2}{2\pi } \int^{2\pi}_0 \frac{h(\phi)}{a^2 +r^2 -2ar\cos ( \theta - \phi)}d\phi
\end{equation*}
This is known as the \emph{Poisson Integral Formula}.
\subsubsection{Wave Equation on a Disk}
Consider the wave equation 
\begin{equation*}
    u_{tt} = c^2 \lap u \twen D_a \subset \R^2
\end{equation*}
with the boundary conditions $u=0$ on $\partial D_a$ where $D_a$ is the disk in $\R^2$ of radius $a$. Assume there exist initial conditions for $u$ and $u_t$. What we are effectively trying to do is solve the eigenvalue problem 
\begin{equation*}
    \lap u + \lambda u =0
\end{equation*}
with the previous boundary conditions. We can try solving separating this in terms of polar coordinates $u(r,\theta) = R(r) \Theta(\theta)$ such that we get 
\begin{equation*}
    \frac{r^2R'' +rR + \lambda r^2R}{R} + \frac{\Theta ''}{\Theta} = 0
\end{equation*}
We can imagine that the $\Theta$ term is equivalent to some constant $-\gamma$ which forces the $R$ term to evaluate to $\gamma$. We therefore have the eigenvalue problem $\Theta''+\gamma \Theta =0$ which we know has the general form $\Theta ( \theta ) = A\cos(n\theta) +B\sin (n\theta)$ and hence $\gamma = n^2$. The $R$ equation is then 
\begin{equation*}
    R'' + \frac{1}{r}R'+ \left( \lambda - \frac{\gamma}{r^2}\right)R = 0
\end{equation*}
We require (as before) that $R(r)$ be bounded as $r\to 0$ and $R(a)=0$. If we make the change of variables $\rho = \sqrt{\lambda}r$ and this changes it to 
\begin{equation*}
    R_{\rho \rho} + \frac{1}{\rho}R_\rho + \left( 1 - \frac{n^2}{\rho^2} \right) R = 0
\end{equation*}
which is known as Bessel's Equation of order $n$. The solutions to this are found via the Frobenius Method and yield Bessel Functions of the First and Second Kind 
\begin{equation*}
    \begin{split}
        J_n(\rho) &= \sum^\infty_{m=0} \frac{(-1)^m}{m! \Gamma (m+n+1)} \left( \frac{\rho}{2} \right)^{2m+n} \text{ Bessel Function of the First Kind} 
    \end{split}
\end{equation*}
For non-integer $n$, the Bessel Function of the second kind is of the form 
\begin{equation*}
    Y_n(\rho)=\frac{J_n(\rho)\cos(n\pi)-J_{-n}(\rho)}{\sin(n\pi)}
\end{equation*}
but for integer values of $n$, it is 
\begin{equation*}
    \begin{split}
        Y_n(\rho) =& - \frac{(\rho/2)^{-n}}{\pi} \sum^{n-1}_{k=0} \frac{(n-k-1)!}{k!} \left( \frac{\rho^2}{4} \right)^k + \frac{2}{\pi} J_n(\rho) \ln \frac{\rho}{2} \\&- \frac{(\rho/2)^n}{\pi} \sum^\infty_{k=0} (\psi (k+1)+\psi (n+k+1)) \frac{(-z^2/4)^k}{k!(n+k)!}
    \end{split}
\end{equation*}
where $\psi(x)$ is the digamma defined as
\begin{equation*}
    \psi (z) = \od{}{z}\ln \Gamma (z) = \frac{\Gamma '(z)}{\Gamma(z)}
\end{equation*}
\section{Fourier Transforms}
\subsection{Distributions}
\begin{definition}
    A \emph{test function} on $\R$ is a smooth function $\phi: \R \to \R$which vanishes outside a finite interval.
\end{definition}
Let $\mathcal{D}$ denote the space of all test functions. 
\begin{definition}
    A distribution on $\R$ is a continous linear map $f:\mathcal{D} \to \R$ via some combination $\phi \mapsto (f,\phi)$.
\end{definition}
$f:\mathcal{D}\to \R$ is continuous if the following holds: $\phi_k, \phi \in \mathcal{D}$ all vanish outside of the same interval and all derivatives $\phi_k^{(i)} \to \phi^{(i)}$ converge uniformly. Then $(f,\phi_k) \to (f,\phi)$ converges uniformly as $k\to \infty$. What's important to note is that distributions are \emph{generalized functions} and cannot be defined in the traditional sense. The rigorous way to denote the use of a distribution is pairing the distribution with a test function which then outputs a number. In other words, it is \emph{functional} which maps from a function space (in this case, the space of all test functions on our interval of choice) to a field (in this case, $\R$). Let us look at some examples 
\begin{example}
    \begin{enumerate}
        \item The Dirac Delta distribution $\delta(x)$ is defined by $(\delta, \phi) = \phi (0) \,\,\, \forall \phi \in \mathcal{D}$.
        \item $f:\phi \mapsto \phi''(0)$
        \item $\displaystyle{f:\phi \mapsto \int^\infty_{-\infty}f(x)\phi(x)dx}$ for a fixed $f$ that has a finite integral.
    \end{enumerate}
\end{example}
This last example is sometimes how we write $(f,\phi)$ (as physicists prefer this because they like to abuse notation) even though $f$ is not a well defined function. If we use this last interpretation and treat $f$ as if it were some smooth function, we can find the derivative of a distribution $f$. We have to define how $(f',\phi)$ maps an arbitrary test function $\phi$ to determine what $f'$ is. We do this via the integral interpretation 
\begin{equation*}
    (f,\phi) = \int^\infty_{-\infty}f'(x)\phi(x)dx = -\int^\infty_{-\infty} f(x) \phi'(x)dx = -(f,\phi)
\end{equation*}
Let us use this to determine the derivative of the Heaviside step function.
\begin{example}
    $H(x)$ is defined by 
    \begin{equation*}
        H(x) = \begin{cases}
            0 & x \leq 0 \\ 1 & x>0
        \end{cases}
    \end{equation*}
    Let us use the definition of the derivative of a distribution 
    \begin{equation*}
        (H',\phi) = -(H,\phi') = -\int^\infty_{-\infty} H(x)\phi'(x)dx = -\int^\infty_0 \phi'(x)dx = \phi(0)
    \end{equation*}
    which is the definition of the Dirac Delta. We can also find the derivative of the Delta function. $(\delta ',\phi) = -(\delta, \phi') =-\phi'(0)$ and so we can show by induction that the $n^{th}$ derivative of the Delta function is defined by $(\delta^{(n)},\phi) = (-1)^n\phi^{(n)}(0)$.
\end{example}
If $f_n$ is a sequence of distributions then $f_n$ weakly converges to a distribution $f$ is $(f_n,\phi)\to (f,\phi)$ as $n\to \infty\,\,\,\forall \phi \in \mathcal{D}$.
The power of distributions allows us to talk about convergence as distributions which is more general. 
\begin{example}
    The Fourier Series of $|x|$ on $[-\pi,\pi]$ is given by 
    \begin{equation*}
        |x| = \frac{\pi}{2} -\sum^\infty_{n=0} \frac{4}{(2n+1)^2 \pi} \cos((2n+1)x)
    \end{equation*}
    Let $S_N$ be the $N^{th}$ partial sum and $S_N \to |x|$ as distributions on $(-\pi,\pi)$. In other words, $(S_N, \phi) \to (|x|,\phi)$ for $\phi \in \mathcal{D}$ that vanish outside of $(-\pi,\pi)$. Let us call $|x| = F(x)$. Then $S_N' \to F'(x)$ as distributions through $(S_n', \phi) = - (S_N,\phi') \to -(F,\phi')=(F',\phi)$. Then we have that 
    \begin{equation*}
        S_N' = \sum ^N_{n=0} \frac{4}{(2n+1)\pi} \sin((2n+1)x)  \to F'(x) = \begin{cases}
            -1 & x\in (-\pi,0) \\
            1 & x\in (0,\pi)
        \end{cases}
    \end{equation*}
    While this makes sense normally, what doesn't make sense is the second derivative which gives utter nonsense. However through the lens of the distributions, it makes perfect sense as 
    \begin{equation*}
        S_N'' = \sum^N_{n=0} \frac{4}{\pi}\cos((2n+1)x) \to F''(x) = 2 \sum^\infty_{n=-\infty} (-1)^n \delta (x-n\pi)
    \end{equation*}
    where we have that 
    \begin{equation*}
        (S_N'',\phi) = \sum^N_{n=0} \int^\pi_{-\pi} \phi(x) \frac{4}{\pi}\cos(nx)dx \to 2\phi(0) \text{ as } N \to \infty
    \end{equation*}
\end{example}

\subsubsection{2D and 3D Distributions}
We use the same idea and define $\mathcal{D}$ to be the test functions on $\R^2$ and $\R^3$ with bounded support. Once again, a distribution is a continuous linear map $f:\mathcal{D} \to \R$. If $f$ is continuous and reasonable function, we can express it via 
\begin{equation*}
    (f,\phi) = \int \!\!\! \int \!\!\! \int_{\R^3} f(x) \phi (x) dV
\end{equation*}
In this way, we can also talk about how distributions can ``solve" differential equations. Suppose we want a distribution $u(x,y)=f(x+y)$ such that it solves the equation $u_x-u_y=0$. Then we want to check that $(u_x-u_y,\phi) = 0 \,\,\, \forall \phi \in \mathcal{D}$. We proceed as standard 
\begin{equation*}
    \begin{split}
        (u_x-u_y,\phi) &= (u_x,\phi)-(u_y,\phi) \\
        &= -(u,\phi_x) + (u,\phi_y) \\
        &= \int\!\!\!\int_{\R^2} u(x,y)(-\phi_x (x,y)+\phi_y(x,y))dA
    \end{split}
\end{equation*}
we now make the change of variables $w=x-y $ and $v= x+y$ so that $\pd{\phi}{w} = 2\phi_w$ and the Jacobian evaluates to $\frac{1}{2}$. Putting this together, we get 
\begin{equation*}
    \begin{split}
        (u_x-u_y,\phi) &= \frac{1}{2}\int \!\!\! \int_{\R^2} f(v) 2\phi_w dw dv \\
        &= \int^\infty_{-\infty} f(v) \int^\infty_{-\infty} \phi_w dw dv 
    \end{split}
\end{equation*}
and since $\phi$ only has support on a finite region, the inner integral evaluates to zero and so therefore any distribution in the form $u(x,y) = f(x+y)$ satisfies this one dimensional transport equation.
