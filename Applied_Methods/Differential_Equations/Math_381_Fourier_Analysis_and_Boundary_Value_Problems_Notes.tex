\section{Partial Differential Equations}
\begin{example}
    Find the function $u(x,y)$ such that $u: \R^2 \mapsto \R $ that satisfies the differential equation $u_{xy}=0$
\end{example}
We can just integrate twice. Once with respect to $x$ and one with respect to $y$. 
\begin{equation*}
    \begin{split}
        u_{xy} &= 0 \\
        \Rightarrow u_x&=c(x) \\
        \Rightarrow u(x,y) &=C(x) + D(y)
    \end{split}
\end{equation*}

\begin{example}
    Find a solution $u(x,y)$ to solve the transport equation 
    \begin{equation*}
        u_x+yu_y=0
    \end{equation*}
\end{example}
Conceptually we can think of this as $\vec{v}\cdot \nabla u = D_{\vec{v}}u=0$ where $\vec{v} = \langle 1,y\rangle$. We effectively want to travel along a path such that the directional derivative in $\vec{v}$ is 0. These paths are called integral curves as they are tangent to the vector field at every point. We now arrive at the characteristic equation 
\begin{equation*}
    \od{y}{x}=\frac{y}{1}
\end{equation*}
since the slope at every point is just rise over run and therefore $\frac{y}{1}$. This leads to the solution 
\begin{equation*}
    y(x) = Ae^{x}
\end{equation*}
where $A$ is found using an initial condition. We note that since the value of the function doesn't depend on where you on each curve but rather \emph{which} curve you're on, we can write $u(x,y) =f(A)$ where $A=ye^{-x}$. Therefore the general solution to the transport equation is 
\begin{equation*}
    u(x,y) = f(ye^{-x})
\end{equation*}
\subsection{Initial and Boundary Conditions}
When we have an $n$-order derivative with respect to one variable in an equation, we then need $n$ conditions in order to uniquely determine a solution to a boundary value problem. For derivatives with respect to time, these are generally deemed initial conditions because they are occur at $t=0$. For variables associated with some sort of domains, these are usually deemed boundary conditions since they occur at the boundary of the system of interest. It is worth noting that some boundary conditions can lead to nonsensical and impossible implications. Let us take the boundary value problem 
\begin{equation*}
    y''+y=x \twen y(0)=y(\pi)=0
\end{equation*}
The solution to the homogeneous equation is $A\cos(x)+B\sin(x)$ and the solution to the particular is $x$. Therefore the general solution is $y_g(x)=A\cos x+B\sin x+x$. Plugging in the boundary values yields to following 
\begin{equation*}
    \begin{split}
        y(0)=0=A\\
        y(\pi)=0=\pi
    \end{split}
\end{equation*}
For PDEs we generally do not have to worry about dimensions of boundary conditions because generally it's a boundary in space and is ``infinite dimensional."
\subsection{Wave Equation}
\subsubsection{Solutions}
\begin{theorem}
    Every solution to the wave equation can be written in the form 
    \begin{equation*}
        u(t,x) = p(\xi)+q(\eta) = p(x-ct)+q(x+ct) \label{theorem:1.2}
    \end{equation*}
\end{theorem}
\begin{proof}
    We change our variables to get 
    \begin{equation*}
        u(t,x) = v(x-ct,x+ct)=v(\xi,\eta)=u\left(\frac{\eta - \xi}{2c}, \frac{\eta+\xi}{2}\right)
    \end{equation*}
    By the chain rule, we get
    \begin{equation}
        \pd{u}{t}=c\left(-\pd{v}{\xi}+\pd{v}{\eta}\right) \hspace{20pt} \pdt{u}{t} = c^2\left(\pdt{v}{\xi} - 2\frac{\partial^2v}{\partial \xi \partial\eta}+\pdt{v}{\eta}\right) \label{eq:1.19}
    \end{equation} % Equation 1.19
    \begin{equation}
        \pd{u}{x} = \pd{v}{\xi}+\pd{v}{\eta} \hspace{20pt} \pdt{u}{x} = \pdt{u}{\xi}+2\frac{\partial^2v}{\partial\xi\partial\eta} + \pdt{v}{\eta} \label{eq:1.20}
    \end{equation} % Equation 1.20
    This implies 
    \begin{equation}
        \square u=-4c^2\frac{\partial^2v}{\partial\xi\partial\eta}=0 \text{ iff } \frac{\partial^2v}{\partial\xi \partial\eta} = 0 \label{eq:1.21}
    \end{equation} % Equation 1.21
    If we create a new variable $w=\pd{v}{\eta}$, we get from the condition in \eqref{eq:1.21} that $\pd{w}{\xi}=0 \therefore w=r(\eta)$. We then get the differential equation 
    \begin{equation*}
        \pd{v}{\eta} = r(\eta) \rightarrow\int\pd{v}{\eta}d\eta = \int r(\eta)d\eta
    \end{equation*}
    By performing the integration with respect to $\eta$, we then get a constant as a function of $\xi$ which leads to the following
    \begin{equation*}
        v(\xi,\eta) - p(\xi)=q(\eta) \rightarrow v(\xi,\eta)=p(\xi)+q(\eta)
    \end{equation*}
\end{proof}

\begin{theorem}[d'Alembert's Formula]
    The solution to the wave equation $u_{tt}=c^2u_{xx}$ with initial conditions $u(0,x)=f(x)$ and $u_t(0,x)=g(x)$ is 
    \begin{equation*}
        u(t,x) = p(\xi)+q(\eta) = \frac{f(\xi)+f(\eta)}{2}+\frac{1}{2c}\int^\eta_\xi g(z)dz 
    \end{equation*}
    where $\xi=x-ct$ and $\eta =x+ct$
\end{theorem}

\begin{proof}
    \begin{equation}
    u(0,x)=p(x-0)+q(x-0)=f(x) \rightarrow p'(x)+q'(x)=f'(x) \label{eq:1.22}
\end{equation} % Equation 1.22
\begin{equation}
    u_t(0,x) = -cp'(x)+cq'(x) = g(x) \rightarrow-p'(x)+q'(x)=\frac{1}{c}g(x) \label{eq:1.23}
\end{equation} % Equation 1.23
If we subtract Equations $\eqref{eq:1.22}$ and $\eqref{eq:1.23}$, we get 
\begin{equation*}
    2p'(x) = f'(x)-\frac{1}{c}g(x) \rightarrow \frac{1}{2}f(x)-\frac{1}{2c}\int^x_0g(z)dz + a 
\end{equation*}
\begin{equation*}
    q(x)=f(x)-p(x)=\frac{1}{2}f(x)+\frac{1}{2c}\int^x_0g(z)dz-a
\end{equation*}
By applying Theorem \ref{theorem:1.2} once again, we reach d'Alembert's solution
\begin{equation}
    u(t,x) = p(\xi)+q(\eta) = \frac{f(\xi)+f(\eta)}{2}+\frac{1}{2c}\int^\eta_\xi g(z)dz \label{eq:dASol}
\end{equation}
\end{proof}



When undergoing external forcing, we have to add an additional forcing term 
\begin{equation*}
    \square u=F(t,x)
\end{equation*}
Let us assume for now that we have homogeneous initial conditions, undergoing the same change of variables above, we get 
\begin{equation*}
    \square u=-4c^2\frac{\partial^2v}{\partial\xi\partial\eta}=F\left(\frac{\eta-\xi}{2c},\frac{\eta+\xi}{2}\right)
\end{equation*}
Integrating both sides with respect to $\eta' /\zeta$ (as a dummy variable) from $\xi$ to $\eta$ gets us
\begin{equation*}
    \pd{v}{\xi}(\xi,\eta)-\pd{v}{\xi}(\xi,\xi) = -\frac{1}{4c^2}\int^\eta_\xi F\left(\frac{\zeta-\eta}{2},\frac{\zeta+\eta}{2}\right)d\zeta
\end{equation*}
Rearranging and doing some algebra for the first partial derivatives in Equations \eqref{eq:1.19} and \eqref{eq:1.20}, we get the following
\begin{equation}
    \pd{v}{\xi}(\xi,\eta)=\frac{1}{2}\pd{u}{x}\left(\frac{\eta-\xi}{2},\frac{\eta+\xi}{2}\right)\label{eq:1.25}
\end{equation}
Which if we plug in $\eta=\xi$ and apply the homogeneous initial conditions, we get 
\begin{equation*}
    \pd{v}{\xi}=-\frac{1}{2c}\pd{u}{t}(0,\xi) + \frac{1}{2}\pd{u}{x}(0,\xi) = 0
\end{equation*}
By the same rearrangement and algebra, we get that $v(\eta,\eta)=0$ and so by integrating 1.25 with respect to $\xi'/\chi$ (once again a dummy variable)
\begin{equation}
    -v(\xi,\eta)=-\frac{1}{4c^2}\int^\eta_\xi\int^\eta_\chi F\left(\frac{\eta-\xi}{2},\frac{\eta+\xi}{2}\right)d\zeta d\chi
\end{equation}
We want to change our integration domain and so we take the following change of variables
\begin{equation*}
    \chi=y-cs \text{ and } \zeta=y+cs
\end{equation*}
which then yields the inequality
\begin{equation*}
    x-ct \leq y-cs \leq y+cs \leq x+ct \rightarrow x-c(t-s)\leq y \leq x+c(t-s)    
\end{equation*}
Effectively what we are doing is changing from the characteristic $\chi\zeta$ domain to the physical integration $tx$ domain. $y$ and $s$ just represent dummy variables that we can integrate $F(s,y)$ over the $tx$ plane. The Jacobian results in the relationship $d\chi d\zeta = 2c dsdy$ which leads to the final result of the particular solution to the forced wave equation to be 
\begin{equation*}
    u(t,x) = \frac{1}{2c}\int \! \! \!  \int_DF(s,y)dsdy=\frac{1}{2c}\int^t_0\int^{x+c(t-s)}_{x-c(t-s)}F(s,y)dyds
\end{equation*}
We can now write the general solution to the classical wave equation as the sum of d'Alembert's solution with the solution above to get 
\begin{equation}
    u(t,x)=\frac{1}{2}[f(x-ct)+f(x+ct)] +\frac{1}{2c}\int^{x+ct}_{x-ct}g(y)dy + \frac{1}{2c}\int^t_0\int^{x+c(t-s)}_{x-c(t-s)}F(s,y)dyds
\end{equation}

\subsubsection{Causality and Energy}
Solutions to the wave equation satisfy causality. That is, a change to the initial conditions can only affect the solution $u(t,x)$ for $t>0$ where $|x-x_0|\leq ct$. In other words, the information only propagates at speed $c$.\\\\
We define the \emph{energy} of a solution $u(t,x)$ as 
\begin{equation*}
    E=\frac{1}{2}\int^\infty_{-\infty}u_t^2+c^2u_x^2dx
\end{equation*}
Let us impose the localizing conditions $u(t,x)\rightarrow 0$ as $x\rightarrow \pm \infty$. If we take the time derivative and differentiate under the integral sign
\begin{equation*}
    \begin{split}
        \od{E}{t} &=\int^\infty_{-\infty}u_{tt}u_t+c^2u_xu_{xt}dx \\
        &=\int^\infty_{-\infty} u_tc^2u_{xx}+c^2u_xu_{xt}dx \\
        &=c^2\int^\infty_{-\infty}-u_xu_{xt}+u_xu_{xt}dx \\
        &=0
    \end{split}
\end{equation*}
where we integrated by parts in the middle. This shows that energy is conserved within solutions to the wave equation. 

\subsection{Heat Equation}
\subsubsection{The Maximum Principle}
\begin{theorem}[The Weak Maximum Principle]
    Let $\gamma >0$. Suppose $u(t,x)$ is a solution to the forced heat equation 
    \begin{equation*}
        \pd{u}{t} = \gamma \pdt{u}{x} + F(t,x)
    \end{equation*}
    on the rectangular domain 
    \begin{equation*}
        R=\{a\leq x \leq b, 0 \leq t \leq c \}
    \end{equation*}
    Assuming that the forcing term is nowhere positive, $F(t,x) \leq 0$, then the maximum of $u(t,x)$ on the closed rectangle $\bar{R}$ is attained at $t=0, x=a,$ or $x=b$.
\end{theorem}
\begin{proof}
    Let us first prove the result under the stronger assumption that $F(t,x) = \pd{u}{t} -\gamma \pdt{u}{x} <0$ which implies $\pd{u}{t} < \gamma \pdt{u}{x}$ everywhere in $R$. Suppose there exists a point $(t^\star, x^\star) \in R$ such that $u(t,x)$ is at a local maximum. By definition, its gradient must be 0.
    \begin{equation*}
        \nabla u(t^\star, x^\star ) = \bold{0} \rightarrow u_t(t^\star,x^\star) = u_{x}(t^\star, x^\star) = 0
    \end{equation*}
    Now let us assume a scalar function $h(x) = (t_\star, x)$. $h(x)$ has a maximum at $x=x^\star$ which means by the second derivative test,
    \begin{equation*}
        h''(x^\star) = u_{xx}(t^\star, x^\star) \leq 0
    \end{equation*}
    This is impossible since we showed earlier that $u_t < u_{xx}$ but $u_t = 0$ and $u_{xx} \leq 0$. As such, our initial assumption that there exists a maximum within $R$ is false. Now we need to exclude the possibility of the maximum occurring at the right edge of the rectangle at a point $(t^\star,x^\star) = (c,x^\star)$ where $a<x^\star < b$. If this is the case, then $g(t) =u(t,x^\star)$ is nondecreasing with respect to time at $t=c$ which implies $g'(t) = u_t(t,x^\star) \geq 0$ which is once again with the conditions we established earlier. \\\\
    To generalize our results to the $F(t,x) \leq 0$, we start with a solution to the forced heat equation $u(t,x)$ and add a little perturbation.
    \begin{equation*}
        v(t,x) = u(t,x) + \varepsilon x^2 \text{ where } \varepsilon>0
    \end{equation*}
    By the chain rule, we get the following 
    \begin{equation*}
        \pd{v}{t} = \pd{u}{t} =\gamma \pdt{u}{x}+F(t,x) = \gamma \pdt{v}{x} -2\gamma \varepsilon+F(t,x) = \gamma \pdt{v}{x} +\tilde{F}(t,x)
    \end{equation*}
    Since $F(t,x) \leq 0$ and $\varepsilon>0$, we know that 
    \begin{equation*}
        \tilde{F}(t,x) = F(t,x) -2\gamma\varepsilon <0
    \end{equation*}
    and we can use what we proved previously to state that $v(t,x)$ cannot attain a a maximum anywhere in $R$ or on the right edge of $R$. Now to rigorously show that this applies to $u(t,x)$, let us denote the maximum value of $u(t,x)$ on any of the three sides as $M$. By our previous logic and mimicking the form of $v(t,x)$, we get
    \begin{equation*}
    u(t,x) \leq v(t,x) \leq M+\varepsilon \max \{a^2,b^2 \}.
    \end{equation*}
    Now let $\varepsilon\rightarrow 0^+$ and we have $u(t,x) \leq M$ everywhere in $R$.
    
\end{proof}
This is known as the \textit{Weak Maximum Principle}. The \textit{Strong Maximum Principle} states that given a non-constant solution $u(t,x)$, any non-initial, non-bonudary point $(t,x) \in \hat{R} = \{a<x<b, 0<t\leq c\}$ is strictly less than its maximum initial and boundary values. In math terms, $u(t,x)<M \ \forall(t,x)\in \hat{R}$. Since the Maximum Principle is used to establish Corollary \ref{coro:mM}, the inequalities become strict in there as well. 
\begin{corollary}
    Supposed $u(t,x)$ solves the heat equation $u_t = \gamma u_{xx}$ with $\gamma > 0$ for $a,x<b$ and $0<t<c$. Set 
    \begin{equation*}
        B=\{(0,x) \ | \ a\leq x \leq b\} \cup \{(t,a)\ | \ 0\leq t \leq c\} \cup \{(t,b)\ | \ 0 \leq t \leq c\}
    \end{equation*}
    and let 
    \begin{equation*}
        M = \max \{u(t,x) \ | \ (t,x) \in B\} \hspace{20pt} m=\min \{u(t,x) \ | \ (t,x) \in B \}
    \end{equation*}
    be respectively the maximum and minimum values for the initial and boundary temperatures. Then $m \leq u(t,x) \leq M$ for all $a \leq x \leq b, 0 \leq t \leq c$.\label{coro:mM}
\end{corollary}
\begin{proof}
        The upper bound $u(t,x) \leq M$ follows immediately from the Maximum principle. To establish the lower bound, let us define a new function $\tilde{u}(t,x) = -u(t,x)$ that also solves the heat equation. This satisfies $\tilde{u}(t,x) \leq -m$ on $B$ by the Maximum Principle (in essence what we're doing is flipping the sign such that the minimum now becomes the maximum). This implies that $u(t,x) = -\tilde{u}(t,x) \geq m$.
    \end{proof}
\begin{theorem}[Uniqueness]
    There is at most, one solution to the Dirichlet initial-boundary value problem for the forced heat equation.
\end{theorem}
\begin{proof}
    Suppose $u$ and $\tilde{u}$ are any two solutions to the forced heat equation with the same initial and boundary value conditions. Their difference $v=u-\tilde{u}$ then solves the homogeneous initial-boundary value problem for the unforced heat equation (since the forcing terms cancel out). By the Maximum Principle and Corollary \ref{coro:mM}, we get the maximum and minimum must appear on the boundaries and therefore $m=0\leq v(t,x) \leq M=0$ everywhere which implies $u\equiv \tilde{u}$ and therefore $u$ is unique.
\end{proof}
We can also use an energy method to determine the uniqueness of the solutions. Suppose $u,v$ solve the heat equation and $u=v$ on $\partial R$. We can define a $w=u-v$ and then $w=0$ on $\partial R$. We now define some quantity 
\begin{equation*}
    E(t)=\int^L_0\frac{1}{2}w^2(x,t)dt
\end{equation*}
If we take time derivatives, we get 
\begin{equation*}
    \begin{split}
        \od{E}{t} &=\int^L_0ww_tdt \\
        &=\int^L_0 wkw_{xx}dx \\
        &=kww_x\bigg|^L_0 - k\int^L_0w_x^2dx\\
        &=-k\int^L_0w_x^2dx
    \end{split}
\end{equation*}
If we set the time equal to zero, we see that hte energy is 
\begin{equation*}
    E(0)=\int^L_0\frac{1}{2}w(x,0)^2dx =0
\end{equation*}
by the fact that $w=0$ on $\partial R$. However, $E(t)$ is non-increasing for $t\in [0,T]$ and since energy cannot be negative, $w(x,t)=0$ in $R$. 
\subsubsection{The Fundamental Solution to the Heat Equation}
Suppose $u(t,x)$ solves the heat equation. Some special properties of these solutions are that they are space and time invariant, there are some special symmetry and similarity solutions, its derivatives solve the heat equation, and linear combinations also solve the heat equation. We therefore seek a solution $Q(t,x)$ with initial condition $Q(0,x)=H(x)$. We choose this form of a solution since $Q(0,ax)=Q(0,x)\,\,\, \forall a>0$. Let us say $Q(x,t)=g(p) $ where $p=\frac{x}{\sqrt{4kt}}$. Therefore 
\begin{equation*}
    \begin{split}
        Q_t &=-\frac{x}{2\sqrt{4k}}xt^{-3/2}g'(p) \\
        Q_{xx}&= g''(p)\left(\frac{1}{\sqrt{4k}}t^{-1/2}\right)^2
    \end{split}
\end{equation*}
and therefore the heat equation turns into 
\begin{equation*}
    -g'(p)\frac{x}{2\sqrt{4k}}xt^{-3/2} = kg''(p)\frac{1}{4k}t^{-1}
\end{equation*}
and therefore we get 
\begin{equation*}
    g''(p)+g'(p)2p=0
\end{equation*}
If we let $h(p)=g'(p)$, we can see that this is a linear first order differential equation and we using an integrating factor $e^{p^2}$. This leads us to get that 
\begin{equation*}
    h=g'(p)=Ae^{-p^2}
\end{equation*}
Therefore 
\begin{equation}
    Q(t,x)=g(p)=A\int^p_{-\infty} e^{-q^2}dq+B
\end{equation}
If $x>0$ and $t\rightarrow 0^+$, then $Q(t,x)$ we have that 
\begin{equation}
    Q(t,x)=A\sqrt{\pi}=1 \to A=\frac{1}{\sqrt\pi}
\end{equation}
If we have some initial condition $\phi (x)$, we can impose some localizing boundary conditions as $x\to \pm \infty$. We can write $\phi(x)$ in the following form 
\begin{equation*}
    \begin{split}
        \phi(x)&=\int^x_{-\infty}\phi'(y)dy \\&= \int^\infty_{-\infty}Q(0,x-y) \phi '(y)dy \\
        &= - \int\phi(y)Q_x(0,x-y) (-1)dy \\
        &=\int^\infty_{-\infty}\phi (y)Q_x(0,x-y)dy
    \end{split}
\end{equation*}
where the $Q_x$ derivative implies with respect to the second entry and not $x$ explicitly inside of the argument. From above, we can wee that 
\begin{equation*}
    Q_x(t,x) = \frac{1}{\sqrt {4\pi kt}}e^{-x^2/4kt}
\end{equation*}
and therefore 
\begin{equation*}
    \phi (x)=\frac{1}{\sqrt{4\pi kt}}\int^\infty_{-\infty} \phi(y)e^{-\frac{(x-y)^2}{4kt}}dy
\end{equation*}
\section{Separation of Variables}
Let us suppose we are given the equation 
\begin{equation*}
    u_t=ku_{xx} \twen x\in [0,l] \twen t\geq 0
\end{equation*}
with $u(0,t)=u(l,t)=0 \,\,\, \forall t$ and $u(0,x)=\phi(x)$. We start with the ansatz $u(t,x)=T(t)X(x)$. Substituting this into the heat equation, we have 
\begin{equation*}
    T'X=kTX'' 
\end{equation*}
and dividing both sides by $XT$, we get 
\begin{equation*}
    \frac{T'}{kT} = \frac{X''}{X} = \lambda
\end{equation*}
For the temporal equation, we get that 
\begin{equation*}
    \frac{T'}{kT}=\lambda \to T=Ae^{\lambda kt}
\end{equation*}
and for the spatial we have 
\begin{equation*}
    \frac{X''}{X}=\lambda
\end{equation*}
We see with here that we require $\lambda <0$ because if $\lambda >0$, we would get $\sinh$ and $\cosh$ spatial solutions which can't satisfy the boundary conditions. $\lambda \neq 0$ because we wold get a linear eigenfunction which can't generate an arbitrary function. The boundary conditions also imply that the linear function would be the trivial solution and therefore 0 can't be an eigenvalue because of that. As such, let $\lambda = -\alpha^2$. With these, we get that $X=B\cos(\alpha x)+C\sin(\alpha x)$. From the boundary values, we see that $B=0$ and $\alpha l = n\pi$ and therefore $\alpha = \frac{n\pi}{l}$. Combining everything, we get 
\begin{equation*}
    u(t,x)=\sum^\infty_{n=1}e^{-\alpha_n^2kt}b_n\sin(\alpha_nx)
\end{equation*}
and thus the initial condition 
\begin{equation*}
    \phi(x) = \sum^\infty_{n=1}b_n\sin (\alpha_nx)
\end{equation*}
This is the genesis of Fourier Analysis.

\section{Fourier Series}
We saw in the previous section that any arbitrary function $\phi(x)$ can be represented as 
\begin{equation*}
    \phi (x) = \sum^\infty_{n=1} A_n \sin\left(\frac{n\pi x}{l}\right)
\end{equation*}
We also have a key identity that 
\begin{equation*}
    \int^l_0 \sin \frac{n\pi x}{l} \sin \frac{m\pi x}{l} dx = \frac{l}{2}\delta_{n,m}
\end{equation*}
where $\delta_{n,m}$ is the Kronecker delta and is 0 when $n\neq m$ and 1 when $n=m$. If we take the integral of $\phi$ and its series representation, we get 
\begin{equation*}
    \begin{split}
        \int^l_0 \phi(x)\sin \frac{m\pi x}{l}dx &= \sum^\infty_{n=1}A_n \int^l_0 \sin \frac{n\pi x}{l}\sin \frac{m\pi x}{l}dx\\ &= A_m \frac{l}{2} \\
        \Rightarrow A_m&=\frac{2}{l}\int^l_0 \phi(x)\sin \frac{m\pi x}{l}dx
    \end{split}
\end{equation*}
\subsection{Sturm-Liouville Theory}
Let us consider $A$ to be the differential operator $-\odt{}{x}$ which acts on all smooth function on the interval $(a,b)$. To see if it's a symmetric operator, we want to see if $\langle Af, g\rangle = \langle f,Ag\rangle $. This is analogous to symmetric matrices which we see 
\begin{equation*}
    \begin{split}
        \ip{Mx}{y} &= (Mx)^Ty \\
        &=x^TM^Ty \\
        &=\ip{x}{My}\,\,\, \forall x,y
    \end{split}
\end{equation*}
We start by just computing it using the $L^2$ inner product
\begin{equation*}
    \begin{split}
        \ip{Af}{g}&=\int^b_a -\odt{}{x}fgdx \\
        &=-\od{f}{x}g\Bigg|^b_a +\int^b_a \od{f}{x}\od{g}{x}dx \\
        &=-\od{f}{x}g\Bigg|^b_a +f\od{g}{x} -\int^b_af \odt{}{x}gdx \\
        &=\ip{f}{Ag} + \left[-f'(b)g(b)+f(b)g'(b)+f'(a)g(a)-f(a)g'(a) \right]
    \end{split}
\end{equation*}
where everything in brackets we shall denote as $B$. If we want $A$ to be a symmetric operator, we require that $B=0$. Under various boundary conditions, $B=0$ and these would be homogeneous Dirichlet, homogeneous Neumann, and periodic boundary conditions. 
\begin{definition}
    Boundary conditions are \emph{symmetric} if $B=0$ for all smooth functions $f(x)$ and $g(x)$ on $(a,b)$.
\end{definition}
This means we need 2 linear combinations of boundary conditions that vanish. 
\begin{theorem}
    Suppose we impose symmetric boundary conditions. If $f,g$ are eigenfunctions of some operator $A$ with different eigenvalues,  $\lambda_f \neq \lambda _g$, then $\ip{f}{g} =0$.
\end{theorem}
\begin{proof}
    \begin{equation*}
        \begin{split}
            \ip{Af}{g} &=\ip{\lambda_ff}{g} \\
            &= \lambda_f \ip{f}{g}\\
        \end{split}
    \end{equation*}
    but also $\ip{Af}{g} =\ip{f}{Ag}=\lambda_g\ip{f}{g}$ and therefore we have the equality 
    \begin{equation*}
        \lambda_f \ip{f}{g} = \lambda_g\ip{f}{g}
    \end{equation*}
    Since $\lambda_f\neq \lambda_f$, this equality can only be true if $\ip{f}{g}$. 
\end{proof}
\subsection{Generalized Fourier Series}
Suppose $\phi(x)$ can be written in the form 
\begin{equation*}
    \phi(x) = \sum^\infty_n A_n X_n(x)
\end{equation*}
where $X_n(x)$ are distinct eigenfunctions with distinct eigenvalues. Then 
\begin{equation*}
    \begin{split}
        \ip{\phi}{X_m} &= \ip{ \sum^\infty_n A_nX_n(x)}{X_n} \\
        &=A_m\ip{X_m}{X_m} \\
        \Rightarrow A_m&= \frac{\ip{\phi}{X_m}}{\ip{X_m}{X_m}}
    \end{split}
\end{equation*}
These eigenvaules must also be real. Let us take a look as to why. If $f,g:(a,b)\mapsto \mathbb{C}$, we define the inner product to be 
\begin{equation*}
    \ip{f}{g} =\int^b_af(x)\overline{g(x)}dx
\end{equation*}
We then have $\ip{Af}{g} =\ip{f}{Ag}$ if 
\begin{equation*}
    f'\bar g - f\bar g' \Big|^b_a=0
\end{equation*}
\begin{theorem}
    If we have symmetric boundary conditions, then all eigenvalues are real.
\end{theorem}
\begin{proof}
    If $Af=\lambda f$, then $\ip{Af}{f}=\lambda \ip{f}{f}$ but also $ \ip{Af}{f} =\ip{f}{Af}=\bar \lambda \ip{f}{f}$. Therefore we require that $\lambda =\bar \lambda$ which is only true when $\lambda $ is real.
\end{proof}
If we suppose $X_1,X_2,\ldots$ are pairwise orthogonal eigenfunctions with eigenvalues $\lambda_1,\lambda_2,\ldots$, then we can arrange them in such a way that $\lambda_1 \leq \lambda_2\leq \ldots$. Suppose we have a function $f:(a,b)\mapsto \R$, we can write its generalized Fourier Series as 
\begin{equation*}
    f(x) =\sum^\infty_{n=1} \frac{\ip{f}{X_n}}{\ip{X_n}{X_n}} X_n
\end{equation*}
Our question is, in what sense can this generalized Fourier Series equal $f(x)$? To understand this, we need to understand some notions of convergence.
\begin{definition}[Pointwise Convergence]
    A series $\displaystyle{\sum^\infty_{n=1}g_n(x)}$ converges \emph{pointwise} to $f$ on $(a,b)$ if for all $x\in (a,b)$, we have 
    \begin{equation*}
        \left| f(x)-\sum^N_{n=1}g_n(x) \right| \rightarrow 0 \text{ as } N\to \infty
    \end{equation*}
\end{definition}
\begin{definition}[$L^2$ Convergence]
    A series $\displaystyle{\sum^\infty_{n=1}g_n(x)}$ \emph{converges in $L^2$} (or \emph{mean squared convergence} if 
    \begin{equation*}
        \bigintsss
        ^b_{\!\!\!\!\!\!\!\!a}\,\,\,\left|f(x)-\sum^N_{n=1}g_n(x)\right|^2dx \to 0 \text{ as } N\to \infty
    \end{equation*}
\end{definition}
In a sense, pointwise convergence allows for slow convergence of some points. $L^2$ convergence allows for there to be big errors but over intervals with small measures such that the integral smooths over it.\\\\
If we assume $X_n$ are orthonormal, then we have 
\begin{equation*}
    f = \sum^\infty_{n=1}\ip{f}{X_n}X_n
\end{equation*}
If we let the partial sums be denoted as $S_N(x)=\displaystyle{\sum^N_{n=1}\ip{f}{X_n}{X_n}}$, we then calculate their differences with $f$ as 
\begin{equation*}
    \begin{split}
        ||f-S_n||^2 &= \ip{f-S_N}{f-S_N} \\
        &= \ip{f}{f}-2\ip{f}{S_N}+\ip{S_N}{S_N} \\
        &= ||f||^2-2\sum^N_{n=1}\ip{f}{X_n}^2 + \ip{\sum^N_{n=1}\ip{f}{X_n}X_n}{\sum^N_{k=1}\ip{f}{X_k}X_k} \\
        &= ||f||^2-2\sum^N_{n=1} \ip{f}{X_n}^2+\sum^N_{n=1}\ip{f}{X_n}^2 \\
        &=||f||^2-\sum^N_{n=1}\ip{f}{X_n}^2
    \end{split}
\end{equation*}
There are three important consequences from this calculation:
\begin{enumerate}
    \item Bessel's Inequality
    \begin{equation*}
        \sum^\infty_{n=1} \ip{f}{X_n}^2 \leq ||f||^2
    \end{equation*}
    which effecitvely says this series will always converge since it is bounded above.
    \item Parseval's Equality which states that $S_N(x)$ converges to $f(x)$ in $L^2$ if and only 
    \begin{equation*}
        \sum^\infty_{n=1}\ip{f}{X_n}^2 = ||f||^2
    \end{equation*}
    \item Riemann-Lebesgue Lemma 
    \begin{equation*}
        \lim_{n\to \infty} \ip{f}{X_n} \to 0
    \end{equation*}
\end{enumerate}
These results are under the assumption that $f(x)$ is square integrable. We can state a few theorems.
\begin{theorem}
    $S_N(x)$ converges to $f(x)$ in $L^2$ for any symmetric boundary conditions.
\end{theorem}
\begin{theorem}[Dirichlet's Theorem]
    For classical Fourier series, consider the function periodic function $f:\R \mapsto \R$. Suppose the partial sums are described by 
    \begin{equation}
        S_N(x) = \frac{A_0}{2} + \sum^N_{n=1}(A_n\cos(nx)+B_n\sin (nx))
    \end{equation}
    If $f$ and $f'$ are piecewise continuous functions, then $\displaystyle{\lim_{N\to \infty}S_N(x)}$ converges pointwise for all $x$ and at the discontinuities, we have that 
    \begin{equation*}
        \lim_{N\to\infty}S_N(x)=\frac{1}{2}(f(x_-)+f(x_+))
    \end{equation*}
\end{theorem}
